Future<void> closeApp(WidgetTester tester, WorkbenchBootstrap bootstrap) async {
  await tester.pumpWidget(const SizedBox());
  var closed = false;
  Object? failure;
  bootstrap.disposeAsync().then(
    (_) => closed = true,
    onError: (Object error) {
      failure = error;
      closed = true;
    },
  );
  // Owner cleanup crosses both widget microtasks and real async operations.
  for (var attempt = 0; attempt < 20 && !closed; attempt++) {
    await tester.pump();
    await tester.runAsync(() => Future<void>.delayed(Duration.zero));
  }
  expect(closed, isTrue, reason: 'All owned resources must finish closing');
  expect(failure, isNull);
  await tester.pump();
}
