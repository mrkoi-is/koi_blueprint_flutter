import 'package:flutter/material.dart';
import 'package:koi_ui/theme/koi_theme_tokens.dart';

/// ListTile retains Material's focus, hover, keyboard and disabled behavior.
class KoiSelectableListTile extends StatefulWidget {
  const KoiSelectableListTile({
    required this.title,
    this.subtitle,
    this.leading,
    this.trailing,
    this.selected = false,
    this.enabled = true,
    this.onTap,
    this.focusNode,
    super.key,
  });

  final Widget title;
  final Widget? subtitle;
  final Widget? leading;
  final Widget? trailing;
  final bool selected;
  final bool enabled;
  final VoidCallback? onTap;
  final FocusNode? focusNode;

  @override
  State<KoiSelectableListTile> createState() => _KoiSelectableListTileState();
}

class _KoiSelectableListTileState extends State<KoiSelectableListTile> {
  bool _focused = false;

  @override
  Widget build(BuildContext context) {
    final tokens = KoiThemeTokens.of(context);
    final colors = Theme.of(context).colorScheme;
    return ListTile(
      title: widget.title,
      subtitle: widget.subtitle,
      leading: widget.leading == null
          ? null
          : IconTheme.merge(
              data: IconThemeData(
                size: 18,
                color: widget.enabled
                    ? colors.onSurfaceVariant
                    : colors.onSurface.withValues(alpha: .38),
              ),
              child: widget.leading!,
            ),
      trailing: widget.trailing,
      selected: widget.selected,
      enabled: widget.enabled,
      onTap: widget.enabled ? widget.onTap : null,
      focusNode: widget.focusNode,
      onFocusChange: (value) => setState(() => _focused = value),
      selectedColor: colors.onSurface,
      selectedTileColor: tokens.selectedBackground,
      hoverColor: tokens.hoverBackground,
      focusColor: tokens.hoverBackground,
      minTileHeight: widget.subtitle == null
          ? tokens.listTileHeight
          : (tokens.density == KoiDensity.compact ? 52 : 64),
      minVerticalPadding: 6,
      visualDensity: VisualDensity.standard,
      horizontalTitleGap: 8,
      minLeadingWidth: 18,
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.circular(KoiRadius.row),
        side: BorderSide(
          color: _focused && widget.enabled
              ? colors.primary
              : Colors.transparent,
          width: 2,
        ),
      ),
      contentPadding: EdgeInsets.symmetric(
        horizontal: tokens.listHorizontalPadding,
      ),
    );
  }
}
