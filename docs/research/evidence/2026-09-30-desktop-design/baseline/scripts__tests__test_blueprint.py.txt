import argparse
import contextlib
import importlib.util
import io
import json
import os
from pathlib import Path

import unittest
from unittest.mock import patch

SPEC = importlib.util.spec_from_file_location('blueprint', Path(__file__).resolve().parents[2] / 'blueprint.py')
blueprint = importlib.util.module_from_spec(SPEC)
SPEC.loader.exec_module(blueprint)
REAL_RUN = blueprint.run
REAL_SDK = blueprint.SDK


def write(path, text):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text, encoding='utf-8')


def fixture_members(root, sdk):
    # Controlled fixture response for the SDK's authoritative workspace listing.
    text = (root / 'pubspec.yaml').read_text(encoding='utf-8')
    return [root / value for value in re.findall(r'^  - ([^\n]+)$', text, re.M)]


class BlueprintTest(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory(prefix='blueprint tests 中文 ')
        self.addCleanup(self.temp.cleanup)
        self.root = Path(self.temp.name).resolve()
        self.app = self.root / 'apps/demo'
        write(self.root / 'pubspec.yaml', 'name: example\nworkspace:\n  - apps/demo\n\ndev_dependencies:\n  analyzer: 14.4.0\n')
        write(self.app / 'pubspec.yaml', 'name: demo\nresolution: workspace\n# 用户自己的注释\n')
        self.enterContext(patch.object(blueprint, 'SDK', return_value=argparse.Namespace(flutter='flutter', dart='dart')))
        self.enterContext(patch.object(blueprint, 'members', side_effect=fixture_members))
        self.enterContext(patch.object(blueprint, 'run', side_effect=self.fake_run))
        self.enterContext(contextlib.redirect_stdout(io.StringIO()))

    def feature(self, kind, name=None, dry_run=False):
        blueprint.feature(argparse.Namespace(path=str(self.app), name=name or f'{kind}_thing', kind=kind, dry_run=dry_run))

    def module(self, name='sales_reports', dry_run=False):
        blueprint.module(argparse.Namespace(workspace=str(self.root), name=name, dry_run=dry_run))

    def snapshot(self):
        return {str(path.relative_to(self.root)): path.read_bytes() for path in self.root.rglob('*') if path.is_file()}

    def test_name_and_platform_validation(self):
        for value in ['../bad', 'Bad', 'a-b', 'class', 'enum', 'extends', 'void', 'await', 'con', 'lpt1', '']:
            with self.subTest(value=value), self.assertRaises(blueprint.BlueprintError):
                blueprint.valid_name(value)
        self.assertEqual(blueprint.valid_name('my_app'), 'my_app')
        self.assertEqual(blueprint.platforms('web,ios'), ['web', 'ios'])
        for value in ['web,web', 'web,bad', '', '../web']:
            with self.assertRaises(blueprint.BlueprintError):
                blueprint.platforms(value)

    def test_feature_multiword_identifiers_imports_and_duplicate_protection(self):
        for kind in ['presentation', 'api', 'local']:
            self.feature(kind)
            destination = self.app / f'lib/features/{kind}_thing'
            self.assertTrue((destination / f'{kind}_thing.dart').is_file())
            self.assertTrue(list((self.app / f'test/features/{kind}_thing').rglob('*_test.dart')))
            for dart in destination.rglob('*.dart'):
                text = dart.read_text(encoding='utf-8')
                self.assertNotIn('package:feature_lab/', text)
                self.assertNotIn(f'{kind}_thingController', text)
                for imported in re.findall(r"(?:import|export|part) '([^']+)'", text):
                    if imported.endswith(('.g.dart', '.freezed.dart')) or imported.startswith(('dart:', 'package:')):
                        continue
                    self.assertTrue((dart.parent / imported).is_file(), f'{dart}: {imported}')
            before = self.snapshot()
            with self.assertRaises(blueprint.BlueprintError):
                self.feature(kind)
            self.assertEqual(before, self.snapshot())
        api = (self.app / 'lib/features/api_thing/presentation/providers/api_thing_providers.dart').read_text(encoding='utf-8')
        self.assertIn('apiThingRepositoryProvider', api)
        self.assertIn('_$ApiThingController', api)
        local = (self.app / 'lib/features/local_thing/data/file_local_thing_repository_native.dart').read_text(encoding='utf-8')
        self.assertIn('FileLocalThingRepository', local)
        model = (self.app / 'lib/features/local_thing/presentation/models/local_thing_editor_state.dart').read_text(encoding='utf-8')
        self.assertIn('_LocalThingEditorState', model)

    def test_name_containing_template_name_is_replaced_only_once(self):
        self.feature('api', 'catalog_search')
        text = (self.app / 'lib/features/catalog_search/presentation/providers/catalog_search_providers.dart').read_text(encoding='utf-8')
        self.assertIn('catalogSearchRepositoryProvider', text)
        self.assertNotIn('catalog_searchSearch', text)
        self.assertIn('features/catalog_search/', text)

    def test_presentation_adds_only_required_dependencies(self):
        self.feature('presentation')
        spec = (self.app / 'pubspec.yaml').read_text(encoding='utf-8')
        self.assertIn('flutter:\n    sdk: flutter', spec)
        self.assertIn('flutter_test:\n    sdk: flutter', spec)
        self.assertNotIn('riverpod', spec)
        self.assertNotIn('koi_core', spec)
        self.assertFalse((self.root / 'packages').exists())

    def test_api_adds_generation_deps_and_registers_core_preserving_existing_constraint(self):
        with (self.app / 'pubspec.yaml').open('a', encoding='utf-8') as target:
            target.write('dependencies:\n  flutter_riverpod: ^3.4.0 # chosen by user\n')
        self.feature('api')
        spec = (self.app / 'pubspec.yaml').read_text(encoding='utf-8')
        self.assertIn('flutter_riverpod: ^3.4.0 # chosen by user', spec)
        self.assertIn('# 用户自己的注释', spec)
        for key in ['riverpod_annotation', 'freezed_annotation', 'json_annotation', 'koi_core', 'riverpod_generator', 'freezed', 'json_serializable', 'build_runner']:
            self.assertRegex(spec, rf'(?m)^  {key}:')
        self.assertTrue((self.root / 'packages/koi_core/pubspec.yaml').is_file())
        self.assertEqual((self.root / 'pubspec.yaml').read_text(encoding='utf-8').count('  - packages/koi_core'), 1)
        self.feature('local')
        self.assertEqual((self.root / 'pubspec.yaml').read_text(encoding='utf-8').count('  - packages/koi_core'), 1)
        self.assertNotIn('koi_network', spec)

    def test_registered_nested_support_is_reused_but_unregistered_collision_is_preserved(self):
        nested = self.root / 'vendor/base'
        write(nested / 'pubspec.yaml', 'name: koi_core\nresolution: workspace\n')
        blueprint.add_members(self.root / 'pubspec.yaml', ['vendor/base'])
        self.feature('local')
        self.assertFalse((self.root / 'packages/koi_core').exists())
        self.assertEqual((nested / 'pubspec.yaml').read_text(encoding='utf-8'), 'name: koi_core\nresolution: workspace\n')
        # An unregistered package cannot masquerade as the actual support member.
        root_spec = self.root / 'pubspec.yaml'
        root_spec.write_text(root_spec.read_text(encoding='utf-8').replace('  - vendor/base\n', ''), encoding='utf-8')
        write(self.root / 'packages/koi_core/user.txt', 'keep')
        before = self.snapshot()
        with self.assertRaises(blueprint.BlueprintError):
            self.feature('api')
        self.assertEqual(before, self.snapshot())

    def test_missing_workspace_membership_fails_without_mutation(self):
        before = self.snapshot()
        with patch.object(blueprint, 'members', return_value=[]), self.assertRaises(blueprint.BlueprintError):
            self.feature('api')
        self.assertEqual(before, self.snapshot())

    def test_feature_rolls_back_source_tests_support_and_exact_manifest_bytes(self):
        (self.app / 'pubspec.yaml').write_bytes(b'name: demo\r\nresolution: workspace\r\n# keep bytes\r\n')
        before = self.snapshot()
        real = blueprint.atomic_bytes
        def fail_root_write(path, content):
            if path == self.root / 'pubspec.yaml':
                raise OSError('disk write denied')
            real(path, content)
        with patch.object(blueprint, 'atomic_bytes', side_effect=fail_root_write), self.assertRaises(OSError):
            self.feature('api')
        self.assertEqual(before, self.snapshot())
        self.assertFalse((self.app / 'lib').exists())
        self.assertFalse((self.root / 'packages').exists())
        self.assertFalse(list(self.root.glob('.feature-*')))

    def symlink(self, link, target):
        link.parent.mkdir(parents=True, exist_ok=True)
        try:
            link.symlink_to(target, target_is_directory=target.is_dir())
        except OSError:
            self.skipTest('symlink privileges unavailable')

    def test_symlink_escape_and_in_workspace_redirect_are_rejected(self):
        outside = self.root / 'outside'
        outside.mkdir()
        self.symlink(self.app / 'lib/features', outside)
        with self.assertRaises(blueprint.BlueprintError):
            self.feature('api')
        self.assertFalse(any(outside.iterdir()))

    def test_symlink_manifest_module_parent_and_template_are_rejected(self):
        outside = self.root / 'outside'
        outside.mkdir()
        self.symlink(self.root / 'modules', outside)
        with self.assertRaises(blueprint.BlueprintError):
            self.module()
        self.assertFalse(any(outside.iterdir()))
        template = self.root / 'template'
        template.mkdir()
        write(self.root / 'secret.txt', 'do not copy')
        self.symlink(template / 'linked.txt', self.root / 'secret.txt')
        with self.assertRaises(blueprint.BlueprintError):
            blueprint.copy(template, self.root / 'copied')

    def fake_run(self, command, cwd, capture=False):
        command = [str(item) for item in command]
        cwd = Path(cwd)
        if command[1:2] == ['create']:
            write(Path(command[-1]) / 'lib/main.dart', 'temporary flutter skeleton')
            write(Path(command[-1]) / 'web/runner_identity.txt', command[command.index('--project-name') + 1])
        if command[1:3] == ['pub', 'get']:
            write(cwd / '.dart_tool/package_config.json', json.dumps({'rootUri': cwd.as_uri()}))
        return ''

    def test_sdk_precedence_and_dart_share_one_flutter_installation(self):
        sdk_root = self.root / 'sdk-choice'
        write(sdk_root / '.fvmrc', '{"flutter":"3.47.2"}')
        environment_flutter = self.root / 'env-sdk/bin/flutter'
        project_flutter = sdk_root / '.fvm/flutter_sdk/bin/flutter'
        path_flutter = self.root / 'path-sdk/bin/flutter'
        for path in (environment_flutter, project_flutter, path_flutter):
            write(path, '')
        for selected, expected in ((str(environment_flutter), environment_flutter),
                                   (None, project_flutter)):
            variables = {'PATH': 'original'}
            if selected:
                variables['FLUTTER_BIN'] = selected
            with patch.dict(os.environ, variables, clear=True), \
                 patch.object(blueprint, 'SDK', REAL_SDK), \
                 patch.object(blueprint.shutil, 'which', return_value=str(path_flutter)), \
                 patch.object(blueprint, 'run', return_value='{"frameworkVersion":"3.47.2"}') as run:
                sdk = blueprint.SDK(sdk_root)
                self.assertEqual(sdk.flutter, expected.resolve())
                self.assertEqual(sdk.dart, expected.parent / 'dart')
                self.assertEqual(run.call_args.args[0][0], expected.resolve())
                self.assertEqual(os.environ['PATH'].split(os.pathsep)[0], str(expected.parent))
